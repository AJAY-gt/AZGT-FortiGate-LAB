config system automation-trigger
    edit "WhenVPN1A is up"
        set event-type event-log
        set logid 37138
        config fields
            edit 1
                set name "vpntunnel"
                set value "HUBFG1-VPN1A"
            next
            edit 2
                set name "action"
                set value "tunnel-up"
            next
        end
    next
end
